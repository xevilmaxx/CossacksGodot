func _measure_extents() -> bool:
    _extents.clear()
    var slots: Dictionary = _ents.get("_slot") as Dictionary
    var batches: Dictionary = _ents.get("_batches") as Dictionary
    for f in _fields:
        var fid := int(f["id"])
        var sl: Dictionary = (slots.get(fid, {}) as Dictionary)
        var key := str(sl.get("key", ""))
        var b: Dictionary = (batches.get(key, {}) as Dictionary)
        var ids: Array = b.get("ids", []) as Array
        var xforms: Array = b.get("xforms", []) as Array
        var idx := int(sl.get("index", -1))
        if key != "res|field" or idx < 0 or idx >= ids.size() or idx >= xforms.size():
            return false
        var xf := xforms[idx] as Transform3D
        if xf == null:
            return false
        var mmi: MultiMeshInstance3D = b.get("node") as MultiMeshInstance3D
        var mesh: Mesh = null
        if mmi != null and mmi.multimesh != null:
            mesh = mmi.multimesh.mesh
        if mesh == null:
            return false
        var aabb: AABB = mesh.get_aabb()
        var sx := xf.basis.x.length()
        var sz := xf.basis.z.length()
        if sx <= 0.0 or sz <= 0.0 or aabb.size.x <= 0.0 or aabb.size.z <= 0.0:
            return false
        # World XZ FULL spans of the rendered decal (field basis is identity-rotated,
        # world3d_entities.gd:1551-1593). Directional half-extent is derived in _half_along().
        _extents[fid] = {
            "hx": aabb.size.x * sx, "hz": aabb.size.z * sz,
            "scale": [sx, xf.basis.y.length(), sz],
            "aabb": [aabb.size.x, aabb.size.y, aabb.size.z], "key": key,
        }
    return _extents.size() == _fields.size()


func _half_along(ext: Dictionary, u: Vector2) -> float:
    return 0.5 * (float(ext["hx"]) * absf(u.x) + float(ext["hz"]) * absf(u.y))

                continue
            var d: float = pi.distance_to(_fields[j]["pos"] as Vector2)
            if d < best_d:
                best_d = d
                best_j = j
        if best_j < 0 or best_d > min_d * LATTICE_TOL:
            continue  # not a lattice neighbour (edge/isolated field) — not an adjacency claim
        var pj: Vector2 = _fields[best_j]["pos"] as Vector2
        var u := (pj - pi) / maxf(best_d, 0.0001)
        var ei: Dictionary = _extents[int(_fields[i]["id"])]
        var ej: Dictionary = _extents[int(_fields[best_j]["id"])]
        var gap: float = best_d - (_half_along(ei, u) + _half_along(ej, u))
        if gap > worst_gap:
            worst_gap = gap
        pairs += 1
        if samples.size() < 12:
            samples.append([int(_fields[i]["id"]), int(_fields[best_j]["id"]),
                snapped(best_d, 3), snapped(gap, 3)])
    _m["adjacent_pairs_checked"] = pairs
    _m["worst_edge_gap_m"] = snapped(worst_gap, 4)
    _m["edge_gap_samples"] = samples
    check("adjacent_fields_touch", pairs >= MIN_FIELDS and worst_gap <= EPS_EDGE,
        "pairs=%d worst_gap=%.3f m <= %.3f m (PORT APPROX, no donor A/B)" % [
        pairs, worst_gap, EPS_EDGE], true)


func _assert_no_building_overlap() -> void:
    var buildings: Array = []
    for e in _game.entities:
        if str(e.get("kind", "")) == "building" and not bool(e.get("dead", false)):
            buildings.append(e)
    var worst := INF
    var violations := 0
    var checked := 0
    for f in _fields:
        var fp: Vector2 = f["pos"] as Vector2
        var ef: Dictionary = _extents[int(f["id"])]
        for b in buildings:
            var bp: Vector2 = b.get("pos", Vector2.ZERO)
            var d: float = fp.distance_to(bp)
            if d <= 0.0001:
                violations += 1
                continue
            var u := (bp - fp) / d
            var r_b := maxf(0.25, float(b.get("block_r", (b["def"] as Dictionary).get("radius", 1.0))))
            var gap: float = d - (_half_along(ef, u) + r_b)
            if gap < worst:
                worst = gap
            if gap < -OVERLAP_EPS:
                violations += 1
            checked += 1
    _m["buildings_checked"] = buildings.size()
    _m["field_building_pairs_checked"] = checked
    _m["worst_field_building_gap_m"] = snapped(worst, 4)
    _m["overlap_violations"] = violations
    check("fields_clear_of_buildings", buildings.size() >= 2 and checked > 0 and violations == 0,
        "buildings=%d pairs=%d worst_gap=%.3f m violations=%d (mill+house trap)" % [
        buildings.size(), checked, worst, violations])

func _probe_ghost() -> void:

    # world_pos(): x = sim.x - off.x, z = sim.y - off.y (world3d_fx.gd:809-818).
    _ghost_sim = Vector2(gp.x + off.x, gp.z + off.y)

        (_ghost_sim.distance_to(_cursor) if is_finite(_ghost_sim.x) else -1.0),

func _capture_async(path: String, is_hover: bool) -> void:
    _capturing = true
    var cv := _main.get("capture_view3d") as SubViewport
    if cv == null or not is_instance_valid(cv):
        check("png_mine_hover" if is_hover else "png_mine_placed", false,
            "main.capture_view3d missing (windowed --screenshot boot required)")
        _capturing = false
        return
    cv.render_target_update_mode = SubViewport.UPDATE_ONCE
    var tex := cv.get_texture()
    if tex == null:
        check("png_mine_hover" if is_hover else "png_mine_placed", false, "null viewport texture")
        _capturing = false
        return
    await RenderingServer.frame_post_draw
    if _finished:
        return
    var img: Image = tex.get_image()
    if img == null or img.is_empty():
        check("png_mine_hover" if is_hover else "png_mine_placed", false,
            "blank SubViewport image — no CPU fallback (main.gd:1906-1913)")
        _capturing = false
        return
    var dir := path.get_base_dir()
    if dir != "" and not DirAccess.dir_exists_absolute(dir):
        DirAccess.make_dir_recursive_absolute(dir)
    var err := img.save_png(path)
    var nb := 0
    var n := 0
    for y in range(0, img.get_height(), 4):
        for x in range(0, img.get_width(), 4):
            var c := img.get_pixel(x, y)
            if c.r > 8.0 / 255.0 or c.g > 8.0 / 255.0 or c.b > 8.0 / 255.0:
                nb += 1
            n += 1
    var frac := float(nb) / float(maxi(n, 1))
    var key := "png_hover_metrics" if is_hover else "png_placed_metrics"
    _m[key] = {"path": path, "w": img.get_width(), "h": img.get_height(),
        "non_black_fraction": snapped(frac, 4), "err": err}
    if is_hover:
        _img_hover = img
    else:
        _img_placed = img
    check("png_mine_hover" if is_hover else "png_mine_placed",
        err == OK and img.get_width() == 1280 and img.get_height() == 720 and frac > 0.5,
        "saved real windowed GPU frame %dx%d err=%d non_black=%.3f" % [
        img.get_width(), img.get_height(), err, frac])
    _capturing = false


func _process(_dt: float) -> bool:
    if _finished or _capturing:
        return false
    _stage_frames += 1
    match _stage:
        0:
            if _stage_frames >= 4:
                if not _configure():
                    _write_result("ABORTED_SETUP")
                else:
                    _goto(1)
        1:
            if _stage_frames >= 3:
                _apply_hover()
                _assert_hover_snap()
                _capture_async(_hover_png, true)
                _goto(2)
        2:
            if _stage_frames >= 3:
                _assert_placement_snap()
                _capture_async(_placed_png, false)
                _goto(3)
        3:
            if _stage_frames >= 3:
                _write_result("PASS" if _ok else "FAIL")
    return false
